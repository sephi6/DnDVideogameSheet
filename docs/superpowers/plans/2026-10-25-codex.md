# Codex Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn section 7 (Journal) into a shared, illustrated, live codex of NPCs, places, factions, creatures and lore — a wall of gold-framed cards with a parchment page per entry — keeping the sheet's roleplay fields in a private Background tab.

**Architecture:** Codex entries are their own entity in one Supabase table (`codex_entries`, or one `localStorage` key in local mode) behind a `CodexStorage` interface, mirroring the quest log. A zustand store changes state first, writes each entry serially, and merges Realtime changes. Images are a single `image` string: a Storage path (cloud upload), a repo gallery path, or a data-url (local upload). The UI is a tab menu, a card wall and a sliding parchment page / editor, built on the Azeroth tokens.

**Tech Stack:** React 18 + TypeScript strict (Vite), zustand 5, framer-motion 11, `@supabase/supabase-js` v2 (Postgres, Realtime `postgres_changes`, Storage), plain CSS, Python 3 + Pillow (art post-processing only).

**Spec:** `docs/superpowers/specs/2026-10-25-codex-design.md` (read it first). Visual reference: `docs/superpowers/mockups/2026-10-25-codex.html`, variant **B · Collection** (open through the dev server at `http://localhost:5173/docs/superpowers/mockups/2026-10-25-codex.html`).

## Review Focus

1. **Upload, then Cancel or upload again** must not leave orphan files in Storage nor delete the image the saved entry still uses. Pinned by the `imagesToDiscard` assertions in Task 2 and their use in the editor (Task 8, step 4).
2. **An entry linked from others gets deleted** (here or on another screen): its chips vanish from every page, nothing crashes, and the dangling id is dropped on the next save. Pinned by the `resolveRelated` / `cleanRelated` / `backlinks` assertions in Task 2 and `saveEntry` in Task 5.
3. **Keys while typing or while the page is open**: typing `a`, `d`, `[`, `]`, `w`, `s`, digits or Escape in a codex field must never move cards, change tabs or sections, or leave the sheet; Escape closes the page / cancels the editor instead of exiting to the party. Pinned by the capture listener in Task 7 step 6 and the key checks in Task 7 step 9 and Task 8 step 7.
4. **Search with accents and case** ("aldrén", "ALDREN") finds "Aldren the Grey". Pinned by the `fold` / `entriesForTab` assertions in Task 2.
5. **Live changes to the entry on screen**: someone deletes the open entry → the page closes; someone edits the entry you are editing → your draft is kept and your Save wins; a partial Realtime row for an unknown entry triggers a reload instead of inserting half an entry. Pinned by `applyChange` in Task 5 and the two-session check in Task 9 step 6.

```sql
-- ARCANA · codex (shared journal of NPCs, places, factions, creatures, lore)
-- Run after 0001_init.sql and 0002_quests.sql in Supabase → SQL Editor. Safe to run again.
--
-- Same permission model as characters and quests: every authenticated user can
-- read, create, edit and delete every entry; anon has no policy. Images live in
-- the public-read Storage bucket "codex"; only authenticated users write to it.

create table if not exists public.codex_entries (
  id          uuid primary key default gen_random_uuid(),
  kind        text not null check (kind in ('npc', 'place', 'faction', 'creature', 'lore')),
  name        text not null default '',
  subtitle    text not null default '',
  attitude    text check (attitude in ('ally', 'neutral', 'hostile', 'unknown')),
  status      text,
  image       text,
  notes       text not null default '',
  related     uuid[] not null default '{}',
  details     jsonb not null default '{}',
  added_by    uuid references public.characters (id) on delete set null,
  created_by  uuid default auth.uid() references auth.users (id) on delete set null,
  created_at  timestamptz not null default now(),
  updated_at  timestamptz not null default now()
);

create index if not exists codex_entries_kind_idx on public.codex_entries (kind);

-- updated_at (the function comes from 0001_init.sql).
drop trigger if exists codex_entries_touch_updated_at on public.codex_entries;
create trigger codex_entries_touch_updated_at before update on public.codex_entries
  for each row execute function public.touch_updated_at();

-- ── Permissions ────────────────────────────────────────────────────────────
alter table public.codex_entries enable row level security;

drop policy if exists "party reads codex_entries" on public.codex_entries;
drop policy if exists "party creates codex_entries" on public.codex_entries;
drop policy if exists "party edits codex_entries" on public.codex_entries;
drop policy if exists "party deletes codex_entries" on public.codex_entries;
create policy "party reads codex_entries" on public.codex_entries for select to authenticated using (true);
create policy "party creates codex_entries" on public.codex_entries for insert to authenticated with check (true);
create policy "party edits codex_entries" on public.codex_entries for update to authenticated using (true) with check (true);
create policy "party deletes codex_entries" on public.codex_entries for delete to authenticated using (true);

-- ── Realtime ───────────────────────────────────────────────────────────────
-- Deletes must carry the whole old row, or the client cannot tell which entry went away.
alter table public.codex_entries replica identity full;

do $$
begin
  if not exists (
    select 1 from pg_publication_tables
    where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = 'codex_entries'
  ) then
    alter publication supabase_realtime add table public.codex_entries;
  end if;
end $$;

-- ── Storage: codex images ──────────────────────────────────────────────────
-- Public read (object names are random UUIDs), so images load with a plain URL
-- and cache well. Writes need a signed-in user.
insert into storage.buckets (id, name, public)
values ('codex', 'codex', true)
on conflict (id) do nothing;

drop policy if exists "party uploads codex images" on storage.objects;
drop policy if exists "party replaces codex images" on storage.objects;
drop policy if exists "party deletes codex images" on storage.objects;
create policy "party uploads codex images" on storage.objects
  for insert to authenticated with check (bucket_id = 'codex');
create policy "party replaces codex images" on storage.objects
  for update to authenticated using (bucket_id = 'codex') with check (bucket_id = 'codex');
create policy "party deletes codex images" on storage.objects
  for delete to authenticated using (bucket_id = 'codex');
```

```ts
/** The five kinds of codex entry. Stored as-is in `codex_entries.kind`. */
export type CodexKind = 'npc' | 'place' | 'faction' | 'creature' | 'lore'

/** How the party stands with an NPC, place, faction or creature. */
export type Attitude = 'ally' | 'neutral' | 'hostile' | 'unknown'

/** Kind-specific fields. Only the keys of the entry's kind are kept (sanitizeDetails). */
export interface CodexDetails {
  metAt?: string
  placeType?: string
  firstVisited?: string
  goal?: string
  creatureType?: string
  vulnerabilities?: string[]
  resistances?: string[]
  immunities?: string[]
  loreType?: string
}

/**
 * One shared codex entry. `image` is a Storage path (`codex/<uuid>.webp`), a repo
 * gallery path (`assets/codex/…`), a data-url (local mode) or null.
 * `related` holds outgoing links; incoming ones are computed (backlinks).
 */
export interface CodexEntry {
  id: string
  kind: CodexKind
  name: string
  subtitle: string
  attitude: Attitude | null
  status: string | null
  image: string | null
  notes: string
  related: string[]
  details: CodexDetails
  addedBy: string | null
  createdAt: string
  updatedAt: string
}

/** One entry changed by someone else (Realtime), already mapped to app types. */
export type CodexChange =
  | { type: 'upsert'; row: CodexEntry; inserted: boolean }
  | { type: 'delete'; id: string }
```

```ts
import type { Attitude, CodexDetails, CodexEntry, CodexKind } from '@/types/codex'

/** Everything the UI needs to know about a kind, so no component branches on kind by hand. */
export interface KindInfo {
  kind: CodexKind
  /** Tab label. */
  label: string
  /** Kicker on the page ("NPC", "Place"…). */
  title: string
  /** In "+ New …" and empty states. */
  singular: string
  shape: 'portrait' | 'landscape'
  subtitleLabel: string
  hasAttitude: boolean
  statuses: readonly string[]
  /** Statuses that send the entry to the end of the wall, greyed. */
  settled: readonly string[]
}

export const KINDS: readonly KindInfo[] = [
  { kind: 'npc', label: 'NPCs', title: 'NPC', singular: 'NPC', shape: 'portrait', subtitleLabel: 'Role', hasAttitude: true, statuses: ['Alive', 'Deceased', 'Unknown'], settled: ['Deceased'] },
  { kind: 'place', label: 'Places', title: 'Place', singular: 'place', shape: 'landscape', subtitleLabel: 'Description', hasAttitude: true, statuses: [], settled: [] },
  { kind: 'faction', label: 'Factions', title: 'Faction', singular: 'faction', shape: 'portrait', subtitleLabel: 'Motto', hasAttitude: true, statuses: [], settled: [] },
  { kind: 'creature', label: 'Bestiary', title: 'Creature', singular: 'creature', shape: 'portrait', subtitleLabel: 'Description', hasAttitude: true, statuses: ['Encountered', 'Defeated'], settled: ['Defeated'] },
  { kind: 'lore', label: 'Items & lore', title: 'Lore', singular: 'entry', shape: 'portrait', subtitleLabel: 'Description', hasAttitude: false, statuses: ['Open', 'Solved'], settled: ['Solved'] },

export function kindInfo(kind: CodexKind): KindInfo {
  const info = KINDS.find((k) => k.kind === kind)
  if (!info) throw new Error(`Unknown codex kind: ${kind}`)
  return info
}

export const ATTITUDES: readonly Attitude[] = ['ally', 'neutral', 'hostile', 'unknown']
export const ATTITUDE_LABEL: Record<Attitude, string> = { ally: 'Ally', neutral: 'Neutral', hostile: 'Hostile', unknown: 'Unknown' }
export const PLACE_TYPES = ['City', 'Village', 'Dungeon', 'Region', 'Landmark', 'Building']
export const LORE_TYPES = ['Item', 'Legend', 'Clue', 'Mystery']

/** The Journal menu: the five shared kinds, then the sheet's private Background. */
export type CodexTab = CodexKind | 'background'
export const TAB_ORDER: readonly CodexTab[] = ['npc', 'place', 'faction', 'creature', 'lore', 'background']

export function stepTab(tab: CodexTab, by: -1 | 1): CodexTab {
  const i = TAB_ORDER.indexOf(tab)
  return TAB_ORDER[(i + by + TAB_ORDER.length) % TAB_ORDER.length]
}

/** A blank entry; the store gives it an id on save. */
export function newEntry(kind: CodexKind, addedBy: string | null, now: string): CodexEntry {
  const info = kindInfo(kind)
  return {
    id: '', kind, name: '', subtitle: '',
    attitude: info.hasAttitude ? 'unknown' : null,
    status: info.statuses[0] ?? null,
    image: null, notes: '', related: [], details: {}, addedBy,
    createdAt: now, updatedAt: now,
  }
}

export function isSettled(entry: CodexEntry): boolean {
  return entry.status !== null && kindInfo(entry.kind).settled.includes(entry.status)
}

/** Lower case, no accents, trimmed: "Aldrén " and "ALDREN" compare equal. */
export function fold(text: string): string {
  return text.normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase().trim()
}

const byName = (a: CodexEntry, b: CodexEntry) => fold(a.name).localeCompare(fold(b.name))

/** The wall of one tab: kind, search over name / subtitle / notes, attitude; settled last, then by name. */
export function entriesForTab(entries: CodexEntry[], kind: CodexKind, query = '', attitude: Attitude | null = null): CodexEntry[] {
  const q = fold(query)
  return entries
    .filter((e) => e.kind === kind)
    .filter((e) => !attitude || e.attitude === attitude)
    .filter((e) => !q || fold(`${e.name}\n${e.subtitle}\n${e.notes}`).includes(q))
    .sort((a, b) => Number(isSettled(a)) - Number(isSettled(b)) || byName(a, b))
}

/** Entries that link to `id` ("Mentioned in"), by name. */
export function backlinks(entries: CodexEntry[], id: string): CodexEntry[] {
  return entries.filter((e) => e.id !== id && e.related.includes(id)).sort(byName)
}

/** The existing entries for `ids`, in order, once each; deleted ones are skipped. */
export function resolveRelated(entries: CodexEntry[], ids: string[]): CodexEntry[] {
  const seen = new Set<string>()
  const out: CodexEntry[] = []
  for (const id of ids) {
    const entry = entries.find((e) => e.id === id)
    if (entry && !seen.has(id)) {
      seen.add(id)
      out.push(entry)
    }
  }
  return out
}

/** What is stored on save: existing ids only, never itself, once each. */
export function cleanRelated(entries: CodexEntry[], ids: string[], selfId: string): string[] {
  return resolveRelated(entries, ids).map((e) => e.id).filter((id) => id !== selfId)
}

/** Related-entry picker: name matches across every kind; an empty query lists nothing. */
export function searchEntries(entries: CodexEntry[], query: string, excludeIds: string[], limit = 6): CodexEntry[] {
  const q = fold(query)
  if (!q) return []
  return entries
    .filter((e) => !excludeIds.includes(e.id) && fold(e.name).includes(q))
    .sort(byName)
    .slice(0, limit)
}

const DETAIL_KEYS: Record<CodexKind, (keyof CodexDetails)[]> = {
  npc: ['metAt'],
  place: ['placeType', 'firstVisited'],
  faction: ['goal'],
  creature: ['creatureType', 'vulnerabilities', 'resistances', 'immunities'],
  lore: ['loreType'],
}
const LIST_KEYS = new Set<keyof CodexDetails>(['vulnerabilities', 'resistances', 'immunities'])

/** Keeps the kind's keys only: strings trimmed and non-empty, damage lists valid and unique. */
export function sanitizeDetails(kind: CodexKind, details: CodexDetails): CodexDetails {
  const out: Record<string, string | string[]> = {}
  for (const key of DETAIL_KEYS[kind]) {
    const value = details[key]
    if (value === undefined) continue
    if (LIST_KEYS.has(key)) {
      const list = [...new Set((value as string[]).filter((t) => DAMAGE_TYPES.includes(t)))]
      if (list.length > 0) out[key] = list
    } else if (typeof value === 'string' && value.trim()) {
      out[key] = value.trim()
    }
  }
  return out as CodexDetails
}

const FACT_LABELS: Partial<Record<keyof CodexDetails, string>> = {
  metAt: 'Met at', placeType: 'Type', firstVisited: 'First visited', goal: 'Known goal',
  creatureType: 'Creature type', loreType: 'Type',
}

/** Label / value rows for the page: the subtitle, then the kind's text details. Empty ones are skipped. */
export function entryFacts(entry: CodexEntry): [string, string][] {
  const rows: [string, string][] = []
  if (entry.subtitle.trim()) rows.push([kindInfo(entry.kind).subtitleLabel, entry.subtitle.trim()])
  for (const key of DETAIL_KEYS[entry.kind]) {
    const label = FACT_LABELS[key]
    const value = entry.details[key]
    if (label && typeof value === 'string' && value.trim()) rows.push([label, value.trim()])
  }
  return rows
}

export type ImageSource = 'none' | 'data' | 'asset' | 'storage'

/** Where an entry's image comes from, by prefix. Anything unrecognised counts as none. */
export function imageSource(image: string | null): ImageSource {
  if (!image) return 'none'
  if (image.startsWith('data:image/')) return 'data'
  if (image.startsWith('assets/')) return 'asset'
  if (image.startsWith('codex/')) return 'storage'
  return 'none'
}

/**
 * Storage images an editor session leaves unused: the initial image if it was
 * replaced, and every upload except the one kept. Never the final image.
 */
export function imagesToDiscard(initial: string | null, final: string | null, uploaded: string[]): string[] {
  const out: string[] = []
  for (const image of [initial, ...uploaded]) {
    if (image && image !== final && imageSource(image) === 'storage' && !out.includes(image)) out.push(image)
  }
  return out
}

/** Next selected id along the wall; clamps at both ends; an unknown current selects the first. */
export function moveSelection(ids: string[], current: string | null, by: -1 | 1): string | null {
  if (ids.length === 0) return null
  const i = current ? ids.indexOf(current) : -1
  if (i < 0) return ids[0]
  return ids[Math.max(0, Math.min(ids.length - 1, i + by))]
}
```

```ts
import { downscaleImage } from '@/lib/image'
import { isSupabaseConfigured, requireSupabase } from '@/lib/supabase'
import type { CodexKind } from '@/types/codex'

const BUCKET = 'codex'
const PREFIX = 'codex/'
/** Checked before decoding: a phone photo is a few MB; 20 MB is clearly not one. */
export const MAX_UPLOAD_BYTES = 20 * 1024 * 1024

/** The `src` for an entry's image; the kind's emblem when there is none. */
export function imageUrl(image: string | null, kind: CodexKind): string {
  const fallback = `assets/codex/default-${kind}.svg`
  switch (imageSource(image)) {
    case 'data':
    case 'asset':
      return image as string
    case 'storage':
      if (!isSupabaseConfigured) return fallback
      return requireSupabase().storage.from(BUCKET).getPublicUrl((image as string).slice(PREFIX.length)).data.publicUrl
    default:
      return fallback
  }
}

/**
 * Resizes the file for the kind's card shape, then stores it: uploaded to the
 * `codex` bucket in the cloud (the entry keeps `codex/<uuid>.webp`), or kept as a
 * data-url locally. `preview` is always a data-url, for instant display.
 */
export async function uploadImage(file: File, kind: CodexKind): Promise<{ image: string; preview: string }> {
  if (!file.type.startsWith('image/')) throw new Error('That file is not an image. Choose a PNG, JPEG or WebP.')
  if (file.size > MAX_UPLOAD_BYTES) throw new Error('That image is over 20 MB. Choose a smaller one.')
  const dataUrl = await downscaleImage(file, kindInfo(kind).shape)
  if (!isSupabaseConfigured) return { image: dataUrl, preview: dataUrl }
  const blob = await (await fetch(dataUrl)).blob()
  const name = `${newId()}.${blob.type === 'image/webp' ? 'webp' : 'jpg'}`
  const { error } = await requireSupabase().storage.from(BUCKET).upload(name, blob, {
    contentType: blob.type,
    cacheControl: '31536000',
    upsert: false,
  })
  if (error) throw error
  return { image: `${PREFIX}${name}`, preview: dataUrl }
}

/** Best-effort removal of Storage images nobody uses any more. Failures only log: an orphan file is harmless. */
export function discardImages(images: string[]): void {
  const names = images.filter((i) => imageSource(i) === 'storage').map((i) => i.slice(PREFIX.length))
  if (names.length === 0 || !isSupabaseConfigured) return
  void requireSupabase()
    .storage.from(BUCKET)
    .remove(names)
    .then(({ error }) => {
      if (error) console.warn('[arcana] could not delete codex images', names, error)
    })
}
```

```ts
import { newId } from '@/data/defaults'
import { isSupabaseConfigured, requireSupabase } from '@/lib/supabase'
import type { CodexChange, CodexEntry } from '@/types/codex'

/**
 * Codex persistence: one row per entry, written whole on Save. Local mode keeps
 * every entry under one key and has no live updates.
 */
export interface CodexStorage {
  readonly name: 'local' | 'supabase'
  loadAll(): Promise<CodexEntry[]>
  upsertEntry(entry: CodexEntry): Promise<void>
  deleteEntry(id: string): Promise<void>
  subscribe(onChange: (c: CodexChange) => void, onStatus: (s: 'live' | 'down') => void): () => void

// ── localStorage ────────────────────────────────────────────────────────────

const KEY = 'arcana:codex:v1'

function readLocal(): CodexEntry[] {
  try {
    const parsed = JSON.parse(localStorage.getItem(KEY) ?? '[]') as unknown
    return Array.isArray(parsed) ? (parsed as CodexEntry[]) : []
  } catch {
    return []
  }
}

/** Throws on a full quota: the store marks the entry failed and offers Retry. */
function writeLocal(entries: CodexEntry[]) {
  localStorage.setItem(KEY, JSON.stringify(entries))
}

export const localCodexStorage: CodexStorage = {
  name: 'local',
  async loadAll() {
    return readLocal()
  },
  async upsertEntry(entry) {
    writeLocal([...readLocal().filter((e) => e.id !== entry.id), entry])
  },
  async deleteEntry(id) {
    writeLocal(readLocal().filter((e) => e.id !== id))
  },
  subscribe() {
    return () => {}
  },
}

// ── Supabase ────────────────────────────────────────────────────────────────

interface CodexRow {
  id: string
  kind: CodexEntry['kind']
  name: string
  subtitle: string
  attitude: CodexEntry['attitude']
  status: string | null
  image: string | null
  notes: string
  related: string[]
  details: CodexEntry['details']
  added_by: string | null
  created_at: string
  updated_at?: string
}

/**
 * A Realtime UPDATE may omit unchanged TOASTed columns (long notes, details):
 * those stay undefined so the store keeps its local value.
 */
const fromRow = (r: CodexRow): CodexEntry => ({
  id: r.id, kind: r.kind, name: r.name, subtitle: r.subtitle, attitude: r.attitude,
  status: r.status, image: r.image, notes: r.notes, related: r.related,
  details: r.details, addedBy: r.added_by, createdAt: r.created_at,
  updatedAt: r.updated_at ?? r.created_at,
})

/** created_by and updated_at are left out: the column default and the trigger fill them. */
const toRow = (e: CodexEntry): CodexRow => ({
  id: e.id, kind: e.kind, name: e.name, subtitle: e.subtitle, attitude: e.attitude,
  status: e.status, image: e.image, notes: e.notes, related: e.related,
  details: e.details, added_by: e.addedBy, created_at: e.createdAt,
})

type Payload = RealtimePostgresChangesPayload<CodexRow & Record<string, unknown>>

export const supabaseCodexStorage: CodexStorage = {
  name: 'supabase',

  async loadAll() {
    const { data, error } = await requireSupabase().from('codex_entries').select('*').order('created_at', { ascending: true })
    if (error) throw error
    return (data as CodexRow[]).map(fromRow)
  },

  async upsertEntry(entry) {
    const { error } = await requireSupabase().from('codex_entries').upsert(toRow(entry), { onConflict: 'id' })
    if (error) throw error
  },

  async deleteEntry(id) {
    const { error } = await requireSupabase().from('codex_entries').delete().eq('id', id)
    if (error) throw error
  },

  subscribe(rawOnChange, rawOnStatus) {
    const client = requireSupabase()
    // Cleared on unsubscribe so late events (e.g. CLOSED) never reach the caller.
    let active = true
    const onChange: typeof rawOnChange = (c) => { if (active) rawOnChange(c) }
    const onStatus: typeof rawOnStatus = (s) => { if (active) rawOnStatus(s) }
    const channel = client
      .channel(`arcana-codex-${newId()}`)
      .on('postgres_changes', { event: '*', schema: 'public', table: 'codex_entries' }, (p: Payload) => {
        if (p.eventType === 'DELETE') onChange({ type: 'delete', id: (p.old as CodexRow).id })
        else onChange({ type: 'upsert', row: fromRow(p.new as CodexRow), inserted: p.eventType === 'INSERT' })
      })
      .subscribe((status) => {
        if (status === 'SUBSCRIBED') onStatus('live')
        else if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT' || status === 'CLOSED') onStatus('down')
      })
    return () => {
      active = false
      void client.removeChannel(channel)
    }
  },
}

export const codexStorage: CodexStorage = isSupabaseConfigured ? supabaseCodexStorage : localCodexStorage
```

```ts
type Set_ = (partial: Partial<CodexState> | ((s: CodexState) => Partial<CodexState>)) => void
type Get_ = () => CodexState

/** Writes per entry run one after another, so the last Save is the last write. */
const chains = new Map<string, Promise<void>>()
/** Entries with a write queued or in flight. */
const inflight = new Map<string, number>()
/** Entries whose last write failed; Retry rewrites them from current state. */
const failed = new Set<string>()
/** Entries that got a foreign change while busy: one reload once they settle. */
const stashed = new Set<string>()
/** Bumped on every write start / end and applied change: a load that began earlier must not overwrite that entry. */
const seqs = new Map<string, number>()
const touch = (id: string) => seqs.set(id, (seqs.get(id) ?? 0) + 1)
/** Entries created on this screen: their Realtime INSERT echo is not a discovery. */
const mine = new Set<string>()
/** Bumped by reset(): work started in an older session must not touch the new one. */
let epoch = 0
let unsubscribe: (() => void) | null = null
let hydration: Promise<void> | null = null
let wasDown = false
let hydrateFailed = false
let loading = 0
let reloadAfterLoad = false
let reloadTimer: ReturnType<typeof setTimeout> | null = null

const DOWN_MESSAGE = 'Live codex updates are disconnected. Retry reconnects.'
const FRESH_KEY = 'arcana:codex-fresh'

function readFresh(): string[] {
  try {
    const parsed = JSON.parse(localStorage.getItem(FRESH_KEY) ?? '[]') as unknown
    return Array.isArray(parsed) ? parsed.filter((x): x is string => typeof x === 'string') : []
  } catch {
    return []
  }
}

function writeFresh(ids: string[]) {
  try {
    localStorage.setItem(FRESH_KEY, JSON.stringify(ids))
  } catch {
    // Per-viewer nicety only: losing it just drops the "New" gems.
  }
}

function settled(set: Set_) {
  if (failed.size > 0) set({ sync: 'error', failedCount: failed.size })
  else if (wasDown) set({ sync: 'error', syncError: DOWN_MESSAGE, failedCount: 0 })
  else set({ sync: 'saved', syncError: null, failedCount: 0 })
}

function write(id: string, set: Set_, get: Get_) {
  const at = epoch
  touch(id)
  inflight.set(id, (inflight.get(id) ?? 0) + 1)
  set({ sync: 'saving', syncError: null })
  const next: Promise<void> = (chains.get(id) ?? Promise.resolve())
    .then(() => {
      if (at !== epoch) return
      const entry = get().entries.find((e) => e.id === id)
      return entry ? codexStorage.upsertEntry(entry) : codexStorage.deleteEntry(id)
    })
    .then(() => {
      if (at !== epoch) return
      failed.delete(id)
      settled(set)
    })
    .catch((err) => {
      if (at !== epoch) return
      failed.add(id)
      console.error('[arcana] failed to save a codex entry', id, err)
      set({ sync: 'error', syncError: readableError(err), failedCount: failed.size })
    })
    .finally(() => {
      if (at !== epoch) return
      touch(id)
      const left = (inflight.get(id) ?? 1) - 1
      if (left <= 0) inflight.delete(id)
      else inflight.set(id, left)
      if (chains.get(id) === next) chains.delete(id)
      // A foreign change landed mid-write: ask the database which one it kept.
      if (left <= 0 && !failed.has(id) && stashed.delete(id)) scheduleReload(set)
      if (loading > 0) reloadAfterLoad = true
    })
  chains.set(id, next)
}

function scheduleReload(set: Set_) {
  if (reloadTimer) return
  const at = epoch
  reloadTimer = setTimeout(() => {
    reloadTimer = null
    if (at !== epoch) return
    void load(set, useCodex.getState).catch((err) => console.error('[arcana] could not reload the codex', err))
  }, 0)
}

/** A partial UPDATE (TOASTed columns omitted) keeps the local value of those fields. */
function merge(old: CodexEntry, row: CodexEntry): CodexEntry {
  const defined = Object.fromEntries(Object.entries(row).filter(([, v]) => v !== undefined))
  return { ...old, ...defined } as CodexEntry
}

function applyChange(c: CodexChange, set: Set_, get: Get_) {
  const id = c.type === 'upsert' ? c.row.id : c.id
  if ((inflight.get(id) ?? 0) > 0 || failed.has(id)) {
    stashed.add(id)
    return
  }
  if (c.type === 'delete') {
    touch(id)
    set((s) => ({ entries: s.entries.filter((e) => e.id !== id), fresh: s.fresh.filter((f) => f !== id) }))
    return
  }
  const old = get().entries.find((e) => e.id === id)
  if (!old && Object.values(c.row).some((v) => v === undefined)) {
    scheduleReload(set) // half a row for an entry we do not have: ask for the whole thing
    return
  }
  touch(id)
  const isDiscovery = c.inserted && !old && !mine.has(id)
  set((s) => {
    const fresh = isDiscovery ? [...s.fresh, id] : s.fresh
    if (isDiscovery) writeFresh(fresh)
    return { entries: [...s.entries.filter((e) => e.id !== id), old ? merge(old, c.row) : c.row], fresh }
  })
}

async function load(set: Set_, get: Get_) {
  const at = epoch
  const since = new Map(seqs)
  loading++
  try {
    const incoming = await codexStorage.loadAll()
    if (at !== epoch) return
    const busy = (id: string) =>
      (inflight.get(id) ?? 0) > 0 || failed.has(id) || (seqs.get(id) ?? 0) !== (since.get(id) ?? 0)
    for (const id of [...stashed]) if (!busy(id)) stashed.delete(id)
    const entries = [...incoming.filter((e) => !busy(e.id)), ...get().entries.filter((e) => busy(e.id))]
    const known = new Set(entries.map((e) => e.id))
    set((s) => ({ entries, loaded: true, fresh: s.fresh.filter((id) => known.has(id)) }))
  } finally {
    if (at === epoch) {
      loading--
      if (loading === 0 && reloadAfterLoad) {
        reloadAfterLoad = false
        scheduleReload(set)
      }
    }
  }
}

export const useCodex = create<CodexState>((set, get) => ({
  entries: [],
  loaded: false,
  sync: 'idle',
  syncError: null,
  failedCount: 0,
  fresh: [],

  hydrate() {
    hydration ??= (async () => {
      const at = epoch
      set({ fresh: readFresh() })
      try {
        // Subscribe first, then read: a change committed between the two is not lost.
        unsubscribe?.()
        unsubscribe = codexStorage.subscribe(
          (c) => {
            if (at === epoch) applyChange(c, set, get)
          },
          (status) => {
            if (at !== epoch) return
            if (status === 'down') {
              wasDown = true
              set({ sync: 'error', syncError: DOWN_MESSAGE })
            } else if (wasDown) {
              wasDown = false
              void load(set, get)
                .then(() => { if (at === epoch) settled(set) })
                .catch((err) => { if (at === epoch) set({ sync: 'error', syncError: readableError(err) }) })
            }
          },
        )
        await load(set, get)
        if (at !== epoch) return
        hydrateFailed = false
      } catch (err) {
        if (at !== epoch) return
        hydrateFailed = true
        console.error('[arcana] could not read the codex', err)
        set({ loaded: true, sync: 'error', syncError: readableError(err) })
      } finally {
        if (at === epoch) hydration = null
      }
    })()
    return hydration
  },

  reset() {
    epoch++
    unsubscribe?.()
    unsubscribe = null
    hydration = null
    chains.clear()
    inflight.clear()
    failed.clear()
    stashed.clear()
    seqs.clear()
    mine.clear()
    wasDown = false
    hydrateFailed = false
    loading = 0
    reloadAfterLoad = false
    if (reloadTimer) clearTimeout(reloadTimer)
    reloadTimer = null
    set({ entries: [], loaded: false, sync: 'idle', syncError: null, failedCount: 0, fresh: [] })
  },

  saveEntry(entry) {
    const id = entry.id || newId()
    const now = new Date().toISOString()
    const current = get().entries.find((e) => e.id === id)
    if (!current) mine.add(id)
    const stored: CodexEntry = {
      ...entry,
      id,
      related: cleanRelated(get().entries, entry.related, id),
      details: sanitizeDetails(entry.kind, entry.details),
      createdAt: current?.createdAt ?? (entry.createdAt || now),
      updatedAt: now,
    }
    set((s) => ({ entries: [...s.entries.filter((e) => e.id !== id), stored] }))
    write(id, set, get)
    return id
  },

  deleteEntry(id) {
    const gone = get().entries.find((e) => e.id === id)
    set((s) => ({ entries: s.entries.filter((e) => e.id !== id), fresh: s.fresh.filter((f) => f !== id) }))
    write(id, set, get)
    if (gone?.image) discardImages([gone.image])
  },

  markSeen(id) {
    if (!get().fresh.includes(id)) return
    const fresh = get().fresh.filter((f) => f !== id)
    writeFresh(fresh)
    set({ fresh })
  },

  async seedDemo(roster) {
    const at = epoch
    await get().hydrate() // so "empty" is judged after the load
    if (at !== epoch || get().entries.length > 0) return
    const demo = demoCodex(roster)
    for (const e of demo) mine.add(e.id)
    set((s) => ({ entries: [...s.entries, ...demo], loaded: true }))
    for (const e of demo) write(e.id, set, get)
    await Promise.all(demo.map((e) => chains.get(e.id) ?? Promise.resolve()))
  },

  async retryFailed() {
    if (wasDown || hydrateFailed || !get().loaded) await get().hydrate()
    for (const id of [...failed]) write(id, set, get)
  },
}))
```

```ts
import { type RefObject, useEffect, useLayoutEffect, useState } from 'react'

const FOCUSABLE = 'button:not([disabled]), [href], input:not([type="hidden"]), select, textarea, [tabindex]:not([tabindex="-1"])'

/** While active, Tab and Shift+Tab cycle inside `ref` (a modal page or editor). */
export function useFocusTrap(ref: RefObject<HTMLElement>, active: boolean) {
  useEffect(() => {
    if (!active) return
    const onKey = (e: KeyboardEvent) => {
      const root = ref.current
      if (e.key !== 'Tab' || !root) return
      const items = [...root.querySelectorAll<HTMLElement>(FOCUSABLE)].filter((el) => el.offsetParent !== null)
      if (items.length === 0) return
      const first = items[0]
      const last = items[items.length - 1]
      const current = document.activeElement
      if (!root.contains(current)) {
        e.preventDefault()
        first.focus()
      } else if (e.shiftKey && current === first) {
        e.preventDefault()
        last.focus()
      } else if (!e.shiftKey && current === last) {
        e.preventDefault()
        first.focus()
      }
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [ref, active])
}

/**
 * The on-screen box of the sheet body that contains `anchor`, kept current on
 * resize, while `active`. The body's own scroll is locked meanwhile, so the page
 * laid over it stays put.
 */
export function useSheetBodyRect(anchor: RefObject<HTMLElement>, active: boolean): DOMRect | null {
  const [rect, setRect] = useState<DOMRect | null>(null)
  useLayoutEffect(() => {
    if (!active) {
      setRect(null)
      return
    }
    const body = anchor.current?.closest<HTMLElement>('.sheet-body')
    if (!body) return
    const measure = () => setRect(body.getBoundingClientRect())
    measure()
    const observer = new ResizeObserver(measure)
    observer.observe(body)
    window.addEventListener('resize', measure)
    const overflow = body.style.overflowY
    body.style.overflowY = 'hidden'
    return () => {
      observer.disconnect()
      window.removeEventListener('resize', measure)
      body.style.overflowY = overflow
    }
  }, [anchor, active])
  return rect
}
```

```tsx
  const ids = useMemo(() => list.map((e) => e.id), [list])
  const selected = selectedId && ids.includes(selectedId) ? selectedId : ids[0] ?? null
  const open = openId ? entries.find((e) => e.id === openId) ?? null : null
  const drawerOpen = open !== null
  const rect = useSheetBodyRect(anchor, drawerOpen)

  void characterId

        play('confirm')
        onOpen(selected)
```

```python
"""Processes generated codex art into app-ready WebP files.

Reads every tmp/imagegen_raw/codex/<kind>-<slug>.png, centre-crops it to the
kind's ratio (5:7 portrait, or 16:9 for places), resizes it, compresses it to
WebP under 300 KB in public/assets/codex/, and prints the lines to paste into
src/data/codexGallery.ts.

    python scripts/process-codex-assets.py
"""
from __future__ import annotations

from pathlib import Path

from PIL import Image

ROOT = Path(__file__).resolve().parents[1]
RAW = ROOT / "tmp" / "imagegen_raw" / "codex"
OUT = ROOT / "public" / "assets" / "codex"
KINDS = {"npc", "place", "faction", "creature", "lore"}
MAX_BYTES = 300_000


def center_crop(im: Image.Image, target_ratio: float) -> Image.Image:
    width, height = im.size
    if width / height > target_ratio:
        target_width = round(height * target_ratio)
        left = (width - target_width) // 2
        return im.crop((left, 0, left + target_width, height))
    target_height = round(width / target_ratio)
    top = (height - target_height) // 2
    return im.crop((0, top, width, top + target_height))


def process(source: Path) -> tuple[str, str, int, int]:
    kind = source.stem.split("-", 1)[0]
    if kind not in KINDS or "-" not in source.stem:
        raise SystemExit(f"{source.name}: name it <kind>-<slug>.png with kind in {sorted(KINDS)}")
    size = (1600, 900) if kind == "place" else (1000, 1400)
    image = center_crop(Image.open(source).convert("RGB"), size[0] / size[1])
    image = image.resize(size, Image.Resampling.LANCZOS)
    destination = OUT / f"{source.stem}.webp"
    for quality in range(92, 49, -2):
        image.save(destination, "WEBP", quality=quality, method=6)
        if destination.stat().st_size < MAX_BYTES:
            return kind, source.stem, quality, destination.stat().st_size
    raise SystemExit(f"Could not compress {source.name} below {MAX_BYTES} bytes")


def label(stem: str) -> str:
    return stem.split("-", 1)[1].replace("-", " ").title()


OUT.mkdir(parents=True, exist_ok=True)
sources = sorted(RAW.glob("*.png"))
if not sources:
    raise SystemExit(f"No PNG files in {RAW}")
lines = []
for path in sources:
    kind, stem, quality, size = process(path)
    print(f"{stem}.webp: q={quality}, {size} bytes")
    lines.append(f"  {{ path: 'assets/codex/{stem}.webp', kind: '{kind}', label: '{label(stem)}' }},")
print("\nPaste into CODEX_GALLERY in src/data/codexGallery.ts:\n")
print("\n".join(lines))
```
