# Quest Log Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A shared, live Warcraft-style quest log as section 8 of every sheet, with a tracker of followed objectives on the other sections.

**Architecture:** Quests are their own entity in three Supabase tables (or one `localStorage` key in local mode), behind a `QuestStorage` interface of one-row operations. A zustand store holds them, changes locally first, writes each touched row serially, and merges Realtime changes by id. The UI is a new section plus a tracker component, styled from the approved Claude Design canvas with the existing Azeroth tokens.

**Tech Stack:** React 18 + TypeScript strict (Vite), zustand, framer-motion, `@supabase/supabase-js` v2 (Realtime `postgres_changes`), plain CSS.

**Spec:** `docs/superpowers/specs/2026-10-15-quest-log-design.md` (read it first). Visual reference: the canvas https://claude.ai/artifact/46ZgC7zAjTDd6h8BSNHjqT (boards Quest log, New quest, Tracker, and the three phone boards).

## Review Focus

1. **Rapid +/− on a counter** (five clicks in a second) must end on the last value everywhere — no flicker back from echoes, no out-of-order writes. Pinned by the per-row write chain and in-flight guard in Task 4, step 3.
2. **Signed out / signed in as someone else** must not show the previous session's quests or keep a Realtime channel open. Pinned by `reset()` in Task 4 and the App wiring in Task 5.
3. **A character deleted while it has quests**: no participant row may point at a missing character in local mode, and group quests stay intact. Pinned by `removeCharacter` in Task 4 and the roster hook in Task 5.
4. **Lowering an objective's target below its progress** in the editor must clamp progress, never show `5 / 3`. Pinned by `clampProgress` in Task 2's check script and its use in Task 7's save.
5. **Phone and tablet widths**: the log must be usable one pane at a time and the tracker must not cover inputs. Pinned by Task 6 (single-pane below 1024 px) and Task 8 (card at the end of the body, collapsible).

```sql
create table if not exists public.quests (

  scope             text not null check (scope in ('group', 'personal')),
  status            text not null default 'active' check (status in ('active', 'completed', 'failed')),
  title             text not null default 'New quest',
  description       text not null default '',
  giver             text not null default '',
  location          text not null default '',
  recommended_level int  check (recommended_level between 1 and 20),
  reward_coins      jsonb not null default '{"pp":0,"gp":0,"ep":0,"sp":0,"cp":0}',
  reward_xp         int  not null default 0 check (reward_xp >= 0),
  reward_items      text[] not null default '{}',
  created_by        uuid references auth.users (id) on delete set null,
  created_at        timestamptz not null default now(),
  updated_at        timestamptz not null default now()
);

create table if not exists public.quest_objectives (
  id         uuid primary key default gen_random_uuid(),
  quest_id   uuid not null references public.quests (id) on delete cascade,
  position   int  not null default 0,
  text       text not null default '',
  target     int  not null default 1 check (target between 1 and 999),
  progress   int  not null default 0 check (progress >= 0),
  updated_at timestamptz not null default now()
);

create table if not exists public.quest_participants (
  quest_id     uuid not null references public.quests (id) on delete cascade,
  character_id uuid not null references public.characters (id) on delete cascade,
  joined       boolean not null default true,
  tracked      boolean not null default true,
  updated_at   timestamptz not null default now(),
  primary key (quest_id, character_id)
);

create index if not exists quest_objectives_quest_idx on public.quest_objectives (quest_id);
create index if not exists quest_participants_character_idx on public.quest_participants (character_id);

-- updated_at (the function comes from 0001_init.sql).
drop trigger if exists quests_touch_updated_at on public.quests;
create trigger quests_touch_updated_at before update on public.quests
  for each row execute function public.touch_updated_at();
drop trigger if exists quest_objectives_touch_updated_at on public.quest_objectives;
create trigger quest_objectives_touch_updated_at before update on public.quest_objectives
  for each row execute function public.touch_updated_at();
drop trigger if exists quest_participants_touch_updated_at on public.quest_participants;
create trigger quest_participants_touch_updated_at before update on public.quest_participants
  for each row execute function public.touch_updated_at();

-- ── Permissions ────────────────────────────────────────────────────────────
alter table public.quests             enable row level security;
alter table public.quest_objectives   enable row level security;
alter table public.quest_participants enable row level security;

do $$
declare t text;
begin
  foreach t in array array['quests', 'quest_objectives', 'quest_participants'] loop
    execute format('drop policy if exists "party reads %1$s" on public.%1$I', t);
    execute format('drop policy if exists "party creates %1$s" on public.%1$I', t);
    execute format('drop policy if exists "party edits %1$s" on public.%1$I', t);
    execute format('drop policy if exists "party deletes %1$s" on public.%1$I', t);
    execute format('create policy "party reads %1$s" on public.%1$I for select to authenticated using (true)', t);
    execute format('create policy "party creates %1$s" on public.%1$I for insert to authenticated with check (true)', t);
    execute format('create policy "party edits %1$s" on public.%1$I for update to authenticated using (true) with check (true)', t);
    execute format('create policy "party deletes %1$s" on public.%1$I for delete to authenticated using (true)', t);
  end loop;
end $$;

-- ── Realtime ───────────────────────────────────────────────────────────────
-- Deletes must carry the whole old row, or the client cannot tell which
-- objective or participant went away.
alter table public.quests             replica identity full;
alter table public.quest_objectives   replica identity full;
alter table public.quest_participants replica identity full;

do $$
declare t text;
begin
  foreach t in array array['quests', 'quest_objectives', 'quest_participants'] loop
    if not exists (
      select 1 from pg_publication_tables
      where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = t
    ) then
      execute format('alter publication supabase_realtime add table public.%I', t);
    end if;
  end loop;
end $$;
```

```ts
import type { Character } from '@/types/character'

/** Group: every character of the table. Personal: the characters that joined. */
export type QuestScope = 'group' | 'personal'
export type QuestStatus = 'active' | 'completed' | 'failed'
export type Coins = Character['inventory']['coins']

export interface Quest {
  id: string
  scope: QuestScope
  status: QuestStatus
  title: string
  description: string
  giver: string
  location: string
  recommendedLevel: number | null
  rewardCoins: Coins
  rewardXp: number
  rewardItems: string[]
  createdAt: string
}

/** Progress is shared by everyone on the quest. Done when progress >= target. */
export interface QuestObjective {
  id: string
  questId: string
  position: number
  text: string
  target: number
  progress: number
}

/**
 * Personal quest: `joined` means the character is on it. Group quest: the row
 * only exists to remember `tracked = false` for that character.
 */
export interface QuestParticipant {
  questId: string
  characterId: string
  joined: boolean
  tracked: boolean
}

export interface QuestSnapshot {
  quests: Quest[]
  objectives: QuestObjective[]
  participants: QuestParticipant[]
}

/** One row changed by someone else (Realtime), already mapped to app types. */
export type QuestChange =
  | { table: 'quests'; type: 'upsert'; row: Quest }
  | { table: 'quests'; type: 'delete'; id: string }
  | { table: 'quest_objectives'; type: 'upsert'; row: QuestObjective }
  | { table: 'quest_objectives'; type: 'delete'; id: string }
  | { table: 'quest_participants'; type: 'upsert'; row: QuestParticipant }
  | { table: 'quest_participants'; type: 'delete'; questId: string; characterId: string }
```

```ts
import type { Quest, QuestObjective, QuestParticipant, QuestScope } from '@/types/quest'

export type Difficulty = 'trivial' | 'easy' | 'normal' | 'hard' | 'deadly'
export type QuestTab = 'active' | 'available' | 'finished'

export const DIFFICULTY_LABEL: Record<Difficulty, string> = {
  trivial: 'Trivial', easy: 'Easy', normal: 'Normal', hard: 'Hard', deadly: 'Deadly',
}

/** Warcraft-style colour band from the quest's level against the character's. */
export function difficulty(recommendedLevel: number | null, characterLevel: number): Difficulty {
  if (recommendedLevel == null) return 'normal'
  const diff = recommendedLevel - characterLevel
  if (diff <= -3) return 'trivial'
  if (diff <= -1) return 'easy'
  if (diff <= 1) return 'normal'
  if (diff <= 3) return 'hard'
  return 'deadly'
}

export function participantOf(
  participants: QuestParticipant[], questId: string, characterId: string,
): QuestParticipant | undefined {
  return participants.find((p) => p.questId === questId && p.characterId === characterId)
}

export function isOnQuest(quest: Quest, participants: QuestParticipant[], characterId: string): boolean {
  return quest.scope === 'group' || participantOf(participants, quest.id, characterId)?.joined === true
}

/** Active, on it, and not explicitly untracked. */
export function isTracked(quest: Quest, participants: QuestParticipant[], characterId: string): boolean {
  if (quest.status !== 'active' || !isOnQuest(quest, participants, characterId)) return false
  return participantOf(participants, quest.id, characterId)?.tracked !== false
}

export function objectivesOf(objectives: QuestObjective[], questId: string): QuestObjective[] {
  return objectives.filter((o) => o.questId === questId).sort((a, b) => a.position - b.position)
}

export const isDone = (o: QuestObjective): boolean => o.progress >= o.target

export function questProgress(objectives: QuestObjective[]): { done: number; total: number } {
  return { done: objectives.filter(isDone).length, total: objectives.length }
}

export function clampProgress(n: number, target: number): number {
  return Math.max(0, Math.min(target, Math.round(n)))
}

/** Group quests first, then personal; inside each, oldest first. */
export function questsForTab(
  quests: Quest[], participants: QuestParticipant[], characterId: string, tab: QuestTab,
): Quest[] {
  const on = (q: Quest) => isOnQuest(q, participants, characterId)
  const pick = quests.filter((q) => {
    if (tab === 'active') return q.status === 'active' && on(q)
    if (tab === 'available') return q.status === 'active' && q.scope === 'personal' && !on(q)
    return q.status !== 'active' && on(q)
  })
  return pick.sort((a, b) =>
    a.scope === b.scope ? a.createdAt.localeCompare(b.createdAt) : a.scope === 'group' ? -1 : 1)
}

/** Ids of the characters on a quest, in roster order. */
export function memberIds(quest: Quest, participants: QuestParticipant[], roster: { id: string }[]): string[] {
  return roster.filter((c) => isOnQuest(quest, participants, c.id)).map((c) => c.id)
}

/** A blank quest; the caller assigns the id. */
export function newQuest(scope: QuestScope, now: string): Quest {
  return {
    id: '',
    scope,
    status: 'active',
    title: 'New quest',
    description: '',
    giver: '',
    location: '',
    recommendedLevel: null,
    rewardCoins: { pp: 0, gp: 0, ep: 0, sp: 0, cp: 0 },
    rewardXp: 0,
    rewardItems: [],
    createdAt: now,
  }
}

/** A blank objective; the caller assigns the id. */
export function newObjective(questId: string, position: number): QuestObjective {
  return { id: '', questId, position, text: '', target: 1, progress: 0 }
}
```

```ts
import type { RealtimePostgresChangesPayload } from '@supabase/supabase-js'
import type { Quest, QuestChange, QuestObjective, QuestParticipant, QuestSnapshot } from '@/types/quest'
import { isSupabaseConfigured, requireSupabase } from '@/lib/supabase'

/**
 * Quest persistence: one small operation per row, so two people editing the
 * same quest never overwrite each other's fields. Local mode keeps the whole
 * snapshot under one key and has no live updates.
 */
export interface QuestStorage {
  readonly name: 'local' | 'supabase'
  loadAll(): Promise<QuestSnapshot>
  upsertQuest(q: Quest): Promise<void>
  deleteQuest(id: string): Promise<void>
  upsertObjective(o: QuestObjective): Promise<void>
  deleteObjective(id: string): Promise<void>
  upsertParticipant(p: QuestParticipant): Promise<void>
  deleteParticipant(questId: string, characterId: string): Promise<void>
  subscribe(onChange: (c: QuestChange) => void, onStatus: (s: 'live' | 'down') => void): () => void
}

// ── localStorage ────────────────────────────────────────────────────────────

const KEY = 'arcana:quests:v1'
const EMPTY: QuestSnapshot = { quests: [], objectives: [], participants: [] }

function readLocal(): QuestSnapshot {
  try {
    const raw = localStorage.getItem(KEY)
    if (!raw) return structuredClone(EMPTY)
    const parsed = JSON.parse(raw) as Partial<QuestSnapshot>
    return {
      quests: Array.isArray(parsed.quests) ? parsed.quests : [],
      objectives: Array.isArray(parsed.objectives) ? parsed.objectives : [],
      participants: Array.isArray(parsed.participants) ? parsed.participants : [],
    }
  } catch {
    return structuredClone(EMPTY)
  }
}

function writeLocal(snapshot: QuestSnapshot) {
  localStorage.setItem(KEY, JSON.stringify(snapshot))
}

function upsertBy<T>(list: T[], row: T, same: (a: T) => boolean): T[] {
  const i = list.findIndex(same)
  if (i >= 0) list[i] = row
  else list.push(row)
  return list
}

export const localQuestStorage: QuestStorage = {
  name: 'local',
  async loadAll() {
    return readLocal()
  },
  async upsertQuest(q) {
    const s = readLocal()
    upsertBy(s.quests, q, (x) => x.id === q.id)
    writeLocal(s)
  },
  async deleteQuest(id) {
    const s = readLocal()
    writeLocal({
      quests: s.quests.filter((q) => q.id !== id),
      objectives: s.objectives.filter((o) => o.questId !== id),
      participants: s.participants.filter((p) => p.questId !== id),
    })
  },
  async upsertObjective(o) {
    const s = readLocal()
    upsertBy(s.objectives, o, (x) => x.id === o.id)
    writeLocal(s)
  },
  async deleteObjective(id) {
    const s = readLocal()
    s.objectives = s.objectives.filter((o) => o.id !== id)
    writeLocal(s)
  },
  async upsertParticipant(p) {
    const s = readLocal()
    upsertBy(s.participants, p, (x) => x.questId === p.questId && x.characterId === p.characterId)
    writeLocal(s)
  },
  async deleteParticipant(questId, characterId) {
    const s = readLocal()
    s.participants = s.participants.filter((p) => !(p.questId === questId && p.characterId === characterId))
    writeLocal(s)
  },
  subscribe() {
    return () => {}

interface QuestRow {
  id: string
  scope: Quest['scope']
  status: Quest['status']
  title: string
  description: string
  giver: string
  location: string
  recommended_level: number | null
  reward_coins: Quest['rewardCoins']
  reward_xp: number
  reward_items: string[]
  created_at: string
}
interface ObjectiveRow { id: string; quest_id: string; position: number; text: string; target: number; progress: number }
interface ParticipantRow { quest_id: string; character_id: string; joined: boolean; tracked: boolean }

const questFromRow = (r: QuestRow): Quest => ({
  id: r.id, scope: r.scope, status: r.status, title: r.title, description: r.description,
  giver: r.giver, location: r.location, recommendedLevel: r.recommended_level,
  rewardCoins: { pp: 0, gp: 0, ep: 0, sp: 0, cp: 0, ...r.reward_coins },
  rewardXp: r.reward_xp, rewardItems: r.reward_items ?? [], createdAt: r.created_at,
})
const questToRow = (q: Quest): QuestRow => ({
  id: q.id, scope: q.scope, status: q.status, title: q.title, description: q.description,
  giver: q.giver, location: q.location, recommended_level: q.recommendedLevel,
  reward_coins: q.rewardCoins, reward_xp: q.rewardXp, reward_items: q.rewardItems, created_at: q.createdAt,
})
const objectiveFromRow = (r: ObjectiveRow): QuestObjective => ({
  id: r.id, questId: r.quest_id, position: r.position, text: r.text, target: r.target, progress: r.progress,
})
const objectiveToRow = (o: QuestObjective): ObjectiveRow => ({
  id: o.id, quest_id: o.questId, position: o.position, text: o.text, target: o.target, progress: o.progress,
})
const participantFromRow = (r: ParticipantRow): QuestParticipant => ({
  questId: r.quest_id, characterId: r.character_id, joined: r.joined, tracked: r.tracked,
})
const participantToRow = (p: QuestParticipant): ParticipantRow => ({
  quest_id: p.questId, character_id: p.characterId, joined: p.joined, tracked: p.tracked,
})

type Payload<T extends Record<string, unknown>> = RealtimePostgresChangesPayload<T>

export const supabaseQuestStorage: QuestStorage = {
  name: 'supabase',

  async loadAll() {
    const client = requireSupabase()
    const [quests, objectives, participants] = await Promise.all([
      client.from('quests').select('*').order('created_at', { ascending: true }),
      client.from('quest_objectives').select('*'),
      client.from('quest_participants').select('*'),
    ])
    if (quests.error) throw quests.error
    if (objectives.error) throw objectives.error
    if (participants.error) throw participants.error
    return {
      quests: (quests.data as QuestRow[]).map(questFromRow),
      objectives: (objectives.data as ObjectiveRow[]).map(objectiveFromRow),
      participants: (participants.data as ParticipantRow[]).map(participantFromRow),
    }
  },

  async upsertQuest(q) {
    const client = requireSupabase()
    const { data: auth } = await client.auth.getUser()
    const { error } = await client
      .from('quests')
      .upsert({ ...questToRow(q), created_by: auth.user?.id ?? null }, { onConflict: 'id' })
    if (error) throw error
  },

  async deleteQuest(id) {
    const { error } = await requireSupabase().from('quests').delete().eq('id', id)
    if (error) throw error
  },

  async upsertObjective(o) {
    const { error } = await requireSupabase().from('quest_objectives').upsert(objectiveToRow(o), { onConflict: 'id' })
    if (error) throw error
  },

  async deleteObjective(id) {
    const { error } = await requireSupabase().from('quest_objectives').delete().eq('id', id)
    if (error) throw error
  },

  async upsertParticipant(p) {
    const { error } = await requireSupabase()
      .from('quest_participants')
      .upsert(participantToRow(p), { onConflict: 'quest_id,character_id' })
    if (error) throw error
  },

  async deleteParticipant(questId, characterId) {
    const { error } = await requireSupabase()
      .from('quest_participants')
      .delete()
      .eq('quest_id', questId)
      .eq('character_id', characterId)
    if (error) throw error
  },

  subscribe(onChange, onStatus) {
    const client = requireSupabase()
    const channel = client
      .channel('arcana-quests')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'quests' }, (p: Payload<QuestRow & Record<string, unknown>>) => {
        if (p.eventType === 'DELETE') onChange({ table: 'quests', type: 'delete', id: (p.old as QuestRow).id })
        else onChange({ table: 'quests', type: 'upsert', row: questFromRow(p.new as QuestRow) })
      })
      .on('postgres_changes', { event: '*', schema: 'public', table: 'quest_objectives' }, (p: Payload<ObjectiveRow & Record<string, unknown>>) => {
        if (p.eventType === 'DELETE') onChange({ table: 'quest_objectives', type: 'delete', id: (p.old as ObjectiveRow).id })
        else onChange({ table: 'quest_objectives', type: 'upsert', row: objectiveFromRow(p.new as ObjectiveRow) })
      })
      .on('postgres_changes', { event: '*', schema: 'public', table: 'quest_participants' }, (p: Payload<ParticipantRow & Record<string, unknown>>) => {
        if (p.eventType === 'DELETE') {
          const old = p.old as ParticipantRow
          onChange({ table: 'quest_participants', type: 'delete', questId: old.quest_id, characterId: old.character_id })
        } else onChange({ table: 'quest_participants', type: 'upsert', row: participantFromRow(p.new as ParticipantRow) })
      })
      .subscribe((status) => {
        if (status === 'SUBSCRIBED') onStatus('live')
        else if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT' || status === 'CLOSED') onStatus('down')
      })
    return () => {
      void client.removeChannel(channel)
    }
  },
}

export const questStorage: QuestStorage = isSupabaseConfigured ? supabaseQuestStorage : localQuestStorage
```

```ts
import { create } from 'zustand'
import type { Character } from '@/types/character'
import type { Quest, QuestChange, QuestObjective, QuestParticipant, QuestSnapshot, QuestStatus } from '@/types/quest'
import { demoQuests, newId } from '@/data/defaults'
import { clampProgress, participantOf } from '@/lib/quests'
import { questStorage } from '@/lib/questStorage'
import { readableError } from '@/lib/supabase'
import type { SyncStatus } from '@/store/roster'

interface QuestsState extends QuestSnapshot {
  loaded: boolean
  sync: SyncStatus
  syncError: string | null
  hydrate: () => Promise<void>
  reset: () => void
  /** From the editor: writes the quest, its objectives, and removes dropped ones. */
  saveQuest: (quest: Quest, objectives: QuestObjective[], creatorId: string | null) => string
  setStatus: (questId: string, status: QuestStatus) => void
  deleteQuest: (questId: string) => void
  setProgress: (objectiveId: string, progress: number) => void
  join: (questId: string, characterId: string) => void
  leave: (questId: string, characterId: string) => void
  setTracked: (questId: string, characterId: string, tracked: boolean) => void
  removeCharacter: (characterId: string) => void
  seedDemo: (roster: Character[]) => Promise<void>
  retryFailed: () => Promise<void>
}

type Set_ = (partial: Partial<QuestsState> | ((s: QuestsState) => Partial<QuestsState>)) => void
type Get_ = () => QuestsState

// Row keys: q:<id>, o:<id>, p:<questId>:<characterId>.
const qKey = (id: string) => `q:${id}`
const oKey = (id: string) => `o:${id}`
const pKey = (questId: string, characterId: string) => `p:${questId}:${characterId}`

/** Writes per row run one after another, so the last click is the last write. */
const chains = new Map<string, Promise<void>>()
/** Rows with a write queued or in flight: Realtime echoes for them are ignored. */
const inflight = new Map<string, number>()
/** Rows whose last write failed; Retry rewrites them from current state. */
const failed = new Set<string>()
let unsubscribe: (() => void) | null = null
let hydration: Promise<void> | null = null
let wasDown = false

/** Writes the row as it is NOW in state: upsert if present, delete if gone. */
function rowWrite(key: string, get: Get_): Promise<void> {
  const [kind, a, b] = key.split(':')
  const s = get()
  if (kind === 'q') {
    const q = s.quests.find((x) => x.id === a)
    return q ? questStorage.upsertQuest(q) : questStorage.deleteQuest(a)
  }
  if (kind === 'o') {
    const o = s.objectives.find((x) => x.id === a)
    return o ? questStorage.upsertObjective(o) : questStorage.deleteObjective(a)
  }
  const p = participantOf(s.participants, a, b)
  return p ? questStorage.upsertParticipant(p) : questStorage.deleteParticipant(a, b)
}

function write(key: string, set: Set_, get: Get_) {
  inflight.set(key, (inflight.get(key) ?? 0) + 1)
  set({ sync: 'saving', syncError: null })
  const previous = chains.get(key) ?? Promise.resolve()
  const next = previous
    .then(() => rowWrite(key, get))
    .then(() => {
      failed.delete(key)
      if (failed.size === 0) set({ sync: 'saved', syncError: null })
    })
    .catch((err) => {
      failed.add(key)
      console.error('[arcana] failed to save a quest row', key, err)
      set({ sync: 'error', syncError: readableError(err) })
    })
    .finally(() => {
      const left = (inflight.get(key) ?? 1) - 1
      if (left <= 0) inflight.delete(key)
      else inflight.set(key, left)
      if (chains.get(key) === next) chains.delete(key)
    })
  chains.set(key, next)
}

function changeKey(c: QuestChange): string {
  if (c.table === 'quests') return qKey(c.type === 'upsert' ? c.row.id : c.id)
  if (c.table === 'quest_objectives') return oKey(c.type === 'upsert' ? c.row.id : c.id)
  return c.type === 'upsert' ? pKey(c.row.questId, c.row.characterId) : pKey(c.questId, c.characterId)
}

/** Someone else's change. Ignored while we are writing that same row. */
function applyChange(c: QuestChange, set: Set_) {
  if ((inflight.get(changeKey(c)) ?? 0) > 0) return
  set((s) => {
    if (c.table === 'quests') {
      if (c.type === 'delete') {
        return {
          quests: s.quests.filter((q) => q.id !== c.id),
          objectives: s.objectives.filter((o) => o.questId !== c.id),
          participants: s.participants.filter((p) => p.questId !== c.id),
        }
      }
      return { quests: [...s.quests.filter((q) => q.id !== c.row.id), c.row] }
    }
    if (c.table === 'quest_objectives') {
      if (c.type === 'delete') return { objectives: s.objectives.filter((o) => o.id !== c.id) }
      return { objectives: [...s.objectives.filter((o) => o.id !== c.row.id), c.row] }
    }
    if (c.type === 'delete') {
      return { participants: s.participants.filter((p) => !(p.questId === c.questId && p.characterId === c.characterId)) }
    }
    return {
      participants: [
        ...s.participants.filter((p) => !(p.questId === c.row.questId && p.characterId === c.row.characterId)),
        c.row,
      ],
    }
  })
}

/** Merge by key: rows in `incoming` replace same-key rows in `current`. */
function mergeSnapshot(current: QuestSnapshot, incoming: QuestSnapshot): QuestSnapshot {
  const byId = <T extends { id: string }>(a: T[], b: T[]) => [...a.filter((x) => !b.some((y) => y.id === x.id)), ...b]
  return {
    quests: byId(current.quests, incoming.quests),
    objectives: byId(current.objectives, incoming.objectives),
    participants: [
      ...current.participants.filter((x) => !incoming.participants.some((y) => y.questId === x.questId && y.characterId === x.characterId)),
      ...incoming.participants,
    ],
  }
}

async function load(set: Set_, get: Get_) {
  const snapshot = await questStorage.loadAll()
  set({ ...mergeSnapshot(get(), snapshot), loaded: true })
}

export const useQuests = create<QuestsState>((set, get) => ({
  quests: [],
  objectives: [],
  participants: [],
  loaded: false,
  sync: 'idle',
  syncError: null,

  hydrate() {
    hydration ??= (async () => {
      try {
        await load(set, get)
        unsubscribe?.()
        unsubscribe = questStorage.subscribe(
          (c) => applyChange(c, set),
          (status) => {
            if (status === 'down') {
              wasDown = true
              set({ sync: 'error', syncError: 'Live quest updates are disconnected. Retry reconnects.' })
            } else if (wasDown) {
              wasDown = false
              void load(set, get).then(() => {
                if (failed.size === 0) set({ sync: 'saved', syncError: null })
              })
            }
          },
        )
      } catch (err) {
        console.error('[arcana] could not read the quests', err)
        set({ loaded: true, sync: 'error', syncError: readableError(err) })
      } finally {
        hydration = null
      }
    })()
    return hydration
  },

  reset() {
    unsubscribe?.()
    unsubscribe = null
    chains.clear()
    inflight.clear()
    failed.clear()
    wasDown = false
    set({ quests: [], objectives: [], participants: [], loaded: false, sync: 'idle', syncError: null })
  },

  saveQuest(quest, objectives, creatorId) {
    const isNew = !get().quests.some((q) => q.id === quest.id)
    const stored: Quest = { ...quest, id: quest.id || newId() }
    const kept = objectives.map((o, i) => ({
      ...o,
      id: o.id || newId(),
      questId: stored.id,
      position: i,
      target: Math.max(1, Math.min(999, Math.round(o.target))),
      progress: clampProgress(o.progress, Math.max(1, Math.min(999, Math.round(o.target)))),
    }))
    const dropped = get().objectives.filter((o) => o.questId === stored.id && !kept.some((k) => k.id === o.id))
    set((s) => ({
      quests: [...s.quests.filter((q) => q.id !== stored.id), stored],
      objectives: [...s.objectives.filter((o) => o.questId !== stored.id), ...kept],
    }))
    write(qKey(stored.id), set, get)
    // Objectives and participants wait for the quest row (foreign key).
    void (chains.get(qKey(stored.id)) ?? Promise.resolve()).then(() => {
      for (const o of kept) write(oKey(o.id), set, get)
      for (const o of dropped) write(oKey(o.id), set, get)
      if (isNew && stored.scope === 'personal' && creatorId) get().join(stored.id, creatorId)
    })
    return stored.id
  },

  setStatus(questId, status) {
    set((s) => ({ quests: s.quests.map((q) => (q.id === questId ? { ...q, status } : q)) }))
    write(qKey(questId), set, get)
  },

  deleteQuest(questId) {
    set((s) => ({
      quests: s.quests.filter((q) => q.id !== questId),
      objectives: s.objectives.filter((o) => o.questId !== questId),
      participants: s.participants.filter((p) => p.questId !== questId),
    }))
    write(qKey(questId), set, get) // the database cascades; local storage cascades too
  },

  setProgress(objectiveId, progress) {
    set((s) => ({
      objectives: s.objectives.map((o) => (o.id === objectiveId ? { ...o, progress: clampProgress(progress, o.target) } : o)),
    }))
    write(oKey(objectiveId), set, get)
  },

  join(questId, characterId) {
    const existing = participantOf(get().participants, questId, characterId)
    const row: QuestParticipant = { questId, characterId, joined: true, tracked: existing?.tracked ?? true }
    set((s) => ({
      participants: [...s.participants.filter((p) => !(p.questId === questId && p.characterId === characterId)), row],
    }))
    write(pKey(questId, characterId), set, get)
  },

  leave(questId, characterId) {
    set((s) => ({ participants: s.participants.filter((p) => !(p.questId === questId && p.characterId === characterId)) }))
    write(pKey(questId, characterId), set, get)
  },

  setTracked(questId, characterId, tracked) {
    const quest = get().quests.find((q) => q.id === questId)
    const existing = participantOf(get().participants, questId, characterId)
    const row: QuestParticipant = {
      questId,
      characterId,
      // A group quest's row only carries the tracking preference.
      joined: existing?.joined ?? quest?.scope === 'group',
      tracked,
    }
    set((s) => ({
      participants: [...s.participants.filter((p) => !(p.questId === questId && p.characterId === characterId)), row],
    }))
    write(pKey(questId, characterId), set, get)
  },

  removeCharacter(characterId) {
    const gone = get().participants.filter((p) => p.characterId === characterId)
    set((s) => ({ participants: s.participants.filter((p) => p.characterId !== characterId) }))
    for (const p of gone) write(pKey(p.questId, p.characterId), set, get)
  },

  async seedDemo(roster) {
    if (get().quests.length > 0) return
    const demo = demoQuests(roster)
    set((s) => ({ ...mergeSnapshot(s, demo), loaded: true }))
    for (const q of demo.quests) write(qKey(q.id), set, get)
    await Promise.all(demo.quests.map((q) => chains.get(qKey(q.id)) ?? Promise.resolve()))
    for (const o of demo.objectives) write(oKey(o.id), set, get)
    for (const p of demo.participants) write(pKey(p.questId, p.characterId), set, get)
  },

  async retryFailed() {
    if (wasDown) {
      await get().hydrate()
      return
    }
    const keys = [...failed]
    // Quests first: objectives and participants reference them.
    keys.sort((a, b) => (a.startsWith('q:') ? -1 : 0) - (b.startsWith('q:') ? -1 : 0))
    for (const key of keys) write(key, set, get)
    await Promise.all(keys.map((k) => chains.get(k) ?? Promise.resolve()))
  },
}))
```

Notes for the implementer:
- `write` reads state at execution time, so a queued write always sends the latest value (Review Focus 1).
- `seedDemo` in local mode writes to `localStorage` through the same path.
- If `tsc` complains about the `set` signature type alias, use zustand's `StoreApi<QuestsState>['setState']` and `['getState']` instead of the hand-written `Set_`/`Get_` — same behaviour.

```tsx
const TABS: [QuestTab, string][] = [['active', 'Active'], ['available', 'Available'], ['finished', 'Finished']]
const EMPTY: Record<QuestTab, string> = {
  active: 'No active quests. Create one with + New quest.',
  available: 'No personal quests of other characters to join.',
  finished: 'Nothing finished yet.',
}

export function QuestsSection({ character }: SectionProps) {
  const { quests, objectives, participants, loaded, setTracked } = useQuests()
  const roster = useRoster((s) => s.characters)
  const narrow = useMediaQuery('(max-width: 1023px)')
  const [tab, setTab] = useState<QuestTab>('active')
  const [selectedId, setSelectedId] = useState<string | null>(null)
  const [showPage, setShowPage] = useState(false)

  const list = useMemo(
    () => questsForTab(quests, participants, character.id, tab),
    [quests, participants, character.id, tab],
  )
  const counts = useMemo(
    () => Object.fromEntries(TABS.map(([t]) => [t, questsForTab(quests, participants, character.id, t).length])) as Record<QuestTab, number>,
    [quests, participants, character.id],
  )
  const selected = list.find((q) => q.id === selectedId) ?? list[0] ?? null

  // A quest deleted or moved out of this tab elsewhere: fall back to the first.
  useEffect(() => {
    if (selectedId && !list.some((q) => q.id === selectedId)) setSelectedId(list[0]?.id ?? null)
  }, [list, selectedId])

  // T tracks / untracks the selected active quest.
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (isTyping(e.target) || (e.key !== 't' && e.key !== 'T') || tab !== 'active' || !selected) return
      e.preventDefault()
      play('toggle')
      setTracked(selected.id, character.id, !isTracked(selected, participants, character.id))
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [tab, selected, participants, character.id, setTracked])

  const members = selected
    ? memberIds(selected, participants, roster).map((id) => roster.find((c) => c.id === id)!).filter(Boolean)
    : []

  const listPane = (
    <section className="panel ql-list">
      <div className="ql-tabs" role="tablist">
        {TABS.map(([t, label]) => (
          <button key={t} type="button" role="tab" aria-selected={tab === t} className="ql-tab" data-active={tab === t}
            onClick={() => { play('toggle'); setTab(t); setSelectedId(null); setShowPage(false) }}>
            {label} <b>{counts[t]}</b>
          </button>
        ))}
      </div>
      {!loaded ? <div className="empty">Loading quests…</div>
        : list.length === 0 ? <div className="empty">{EMPTY[tab]}</div>
        : <QuestList quests={list} objectives={objectives} participants={participants} characterId={character.id}
            characterLevel={character.identity.level} selectedId={selected?.id ?? null}
            onSelect={(id) => { setSelectedId(id); setShowPage(true) }} />}
    </section>
  )

  const page = selected && (
    <QuestPage quest={selected} objectives={objectivesOf(objectives, selected.id)} members={members}
      characterId={character.id} characterLevel={character.identity.level} mode={tab} onEdit={() => {}} />
  )

  if (narrow) {
    return showPage && page ? (
      <div>
        <div className="ql-back"><Button variant="small ghost" cue="back" onClick={() => setShowPage(false)}>◂ Quest log</Button></div>
        {page}
      </div>
    ) : listPane
  }
  return <div className="quest-log">{listPane}{page ?? <div className="empty">Select a quest.</div>}</div>
}
```

```tsx
import { newObjective } from '@/lib/quests'
import type { Quest, QuestObjective } from '@/types/quest'

const COINS = [['pp', 'Platinum'], ['gp', 'Gold'], ['ep', 'Electrum'], ['sp', 'Silver'], ['cp', 'Copper']] as const

interface Props {
  initial: Quest
  initialObjectives: QuestObjective[]
  characterName: string
  onSave: (quest: Quest, objectives: QuestObjective[]) => void
  onCancel: () => void
}

export function QuestEditor({ initial, initialObjectives, characterName, onSave, onCancel }: Props) {
  const [quest, setQuest] = useState<Quest>(initial)
  const [objectives, setObjectives] = useState<QuestObjective[]>(
    initialObjectives.length ? initialObjectives : [newObjective(initial.id, 0)],
  )
  const isNew = !initial.id
  const patch = (p: Partial<Quest>) => setQuest((q) => ({ ...q, ...p }))
  const patchObjective = (i: number, p: Partial<QuestObjective>) =>
    setObjectives((list) => list.map((o, j) => (j === i ? { ...o, ...p } : o)))
  const move = (i: number, by: -1 | 1) =>
    setObjectives((list) => {
      const next = [...list]
      const j = i + by
      if (j < 0 || j >= next.length) return list
      ;[next[i], next[j]] = [next[j], next[i]]
      return next
    })
  const canSave = quest.title.trim().length > 0

  return (
    <div className="quest-editor">
      <section className="panel card">
        <header className="row" style={{ justifyContent: 'space-between', alignItems: 'flex-start' }}>
          <h3 className="panel-title" style={{ margin: 0 }}><span>{isNew ? 'New quest' : 'Edit quest'}</span></h3>

            <TextField label="Where" value={quest.location} onChange={(v) => patch({ location: v })} />
            <NumberField label="Rec. level" value={quest.recommendedLevel ?? 0} min={0} max={20}
              onChange={(v) => patch({ recommendedLevel: v > 0 ? v : null })} />
          </div>
          <TextAreaField label="Description" rows={5} value={quest.description} onChange={(v) => patch({ description: v })} />
        </div>
      </section>

      <div className="stack" style={{ gap: 14 }}>
        <section className="panel card">
          <header className="row" style={{ justifyContent: 'space-between', alignItems: 'flex-start' }}>
            <h3 className="panel-title" style={{ margin: 0 }}><span>Objectives</span></h3>
            <div className="row" style={{ gap: 6 }}>
              <Button variant="small ghost" onClick={() => setObjectives((l) => [...l, newObjective(quest.id, l.length)])}>+ Add</Button>
            </div>
          </header>
          <div className="stack" style={{ gap: 8, marginTop: 10 }}>
            {objectives.map((o, i) => (
              <div key={o.id || `new-${i}`} className="qe-objective">
                <input className="input grow" aria-label={`Objective ${i + 1}`} value={o.text} placeholder="What has to be done"
                  onChange={(e) => patchObjective(i, { text: e.target.value })} />

                  <NumberField value={o.target} min={1} max={999} onChange={(v) => patchObjective(i, { target: v })} />
                </div>
                <Button variant="small ghost" onClick={() => move(i, -1)} title="Move up">▴</Button>
                <Button variant="small ghost" onClick={() => move(i, 1)} title="Move down">▾</Button>
                <Button variant="small ghost danger" cue="back" title="Remove objective"
                  onClick={() => setObjectives((l) => l.filter((_, j) => j !== i))}>✕</Button>
              </div>
            ))}
            <p className="muted" style={{ margin: 0, fontSize: 14 }}>A target of 1 is a tick box; more makes a counter (0 / 7).</p>
```

```tsx
import { useState } from 'react'
import { Button } from '@/components/ui/controls'
import { difficulty, isDone, isTracked, objectivesOf } from '@/lib/quests'
import { play } from '@/lib/sfx'
import { useQuests } from '@/store/quests'

interface Props {
  characterId: string
  characterLevel: number
  variant: 'column' | 'card'
  onOpenLog: () => void
}

export function QuestTracker({ characterId, characterLevel, variant, onOpenLog }: Props) {
  const { quests, objectives, participants } = useQuests()
  const [open, setOpen] = useState(false)
  const tracked = quests.filter((q) => isTracked(q, participants, characterId))
  if (tracked.length === 0) return null

  const shown = variant === 'card' && !open ? tracked.slice(0, 1) : tracked
  const body = shown.map((q) => {
    const d = difficulty(q.recommendedLevel, characterLevel)
    return (
      <div key={q.id} className="tk-q">
        <div className="t">
          <span className={`lvn lv-${d}`}>[{q.recommendedLevel ?? '–'}]</span>
          <span className="tt">{q.title}</span>
        </div>
        {objectivesOf(objectives, q.id).map((o) => (
          <div key={o.id} className="tk-o" data-done={isDone(o)}>
            <span>{o.text}</span>
            {o.target > 1 && <span className="n">{o.progress}/{o.target}</span>}
          </div>
        ))}
      </div>
    )
  })

  if (variant === 'column') {
    return (
      <aside className="tracker" aria-label="Tracked quests">
        <div className="tk-head">Tracked<span>{tracked.length}</span></div>
        {body}
        <Button variant="small ghost" onClick={onOpenLog}>Open quest log ▸</Button>
      </aside>
    )
  }
  return (
    <aside className="tracker-card" aria-label="Tracked quests">
      <button type="button" className="tk-toggle" aria-expanded={open}
        onClick={() => { play('toggle'); setOpen((v) => !v) }}>
        <span>Tracked</span>
        <span className="tk-count">{open ? 'Hide ▾' : `${tracked.length > 1 ? `1 of ${tracked.length} ` : ''}▴`}</span>
      </button>
      {body}
      {open && <Button variant="small ghost" onClick={onOpenLog}>Open quest log ▸</Button>}
    </aside>
  )
}
```
